-- Q11: Which country topped each month, and did revenue go up or down?
WITH monthly_revenue AS (
  SELECT c.country,
    TO_CHAR(o.order_date, 'YYYY-MM') AS month,
    SUM(o.total_amount) AS revenue
  FROM customers c
  JOIN orders o ON c.customer_id = o.customer_id
  GROUP BY c.country, TO_CHAR(o.order_date, 'YYYY-MM')
),
ranked AS (
  SELECT country, month, revenue,
    RANK() OVER (PARTITION BY month ORDER BY revenue DESC) AS rnk
  FROM monthly_revenue
)
SELECT month, country, revenue,
  LAG(revenue) OVER (PARTITION BY country ORDER BY month) AS prev,
  CASE
    WHEN LAG(revenue) OVER (...) IS NULL THEN 'first month'
    WHEN revenue > prev THEN 'up ↑'
    ELSE 'down ↓'
  END AS direction
FROM ranked WHERE rnk = 1 ORDER BY month;
